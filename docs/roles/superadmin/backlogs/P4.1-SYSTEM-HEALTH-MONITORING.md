## Overview

Comprehensive system health monitoring dashboard for SuperAdmin with real-time metrics, error tracking, performance monitoring, and alerting.

## Business Value

### Problem Statement
- No visibility into system performance (response times, error rates)
- Database performance issues not detected until users complain
- No alerting for critical errors
- Cannot track system usage patterns for capacity planning
- Downtime incidents discovered reactively, not proactively

### Expected Outcomes
- **Uptime:** Improve from 95% to 99.5% through proactive monitoring
- **Issue Detection:** 90% of issues detected before user reports
- **Performance:** Identify and fix slow queries reducing avg response time by 40%

### Metrics Collection

**Middleware for API Monitoring:**
```typescript
// src/middleware/metrics.ts
import { NextResponse } from 'next/server';
import type { NextRequest } from 'next/server';

const metrics = {
  requestCounts: new Map<string, number>(),
  responseTimes: new Map<string, number[]>(),
  errorCounts: new Map<string, number>(),
};

export async function metricsMiddleware(request: NextRequest) {
  const startTime = Date.now();
  const path = request.nextUrl.pathname;

  try {
    const response = await NextResponse.next();
    const duration = Date.now() - startTime;

    // Record metrics
    recordMetric(path, duration, response.status);

    return response;
  } catch (error) {
    recordError(path, error);
    throw error;
  }
}

function recordMetric(path: string, duration: number, status: number) {
  // Increment request count
  metrics.requestCounts.set(path, (metrics.requestCounts.get(path) || 0) + 1);

  // Record response time
  if (!metrics.responseTimes.has(path)) {
    metrics.responseTimes.set(path, []);
  }
  metrics.responseTimes.get(path)!.push(duration);

  // Record errors (4xx, 5xx)
  if (status >= 400) {
    metrics.errorCounts.set(path, (metrics.errorCounts.get(path) || 0) + 1);
  }
}
```

### Database Schema

```prisma
model SystemMetric {
  id          String   @id @default(cuid())
  timestamp   DateTime @default(now())
  metricType  String   // "request", "error", "database", "memory"
  metricName  String   // Specific metric identifier
  value       Float
  metadata    Json?    // Additional context

  @@index([timestamp, metricType])
}

model SystemError {
  id          String   @id @default(cuid())
  timestamp   DateTime @default(now())
  errorType   String
  message     String
  stackTrace  String?
  context     Json?    // Request details, user, etc.
  resolved    Boolean  @default(false)
  resolvedAt  DateTime?

  @@index([timestamp, errorType])
}

model AlertRule {
  id          String   @id @default(cuid())
  name        String
  condition   Json     // { metric: "errorRate", operator: ">", threshold: 5, duration: 300 }
  enabled     Boolean  @default(true)
  channels    String[] // ["email", "slack"]
  createdBy   String
  createdAt   DateTime @default(now())
}

model AlertHistory {
  id          String   @id @default(cuid())
  ruleId      String
  triggeredAt DateTime @default(now())
  resolvedAt  DateTime?
  acknowledged Boolean  @default(false)
  acknowledgedBy String?

  rule        AlertRule @relation(...)
}
```

### Alert Evaluation (Background Job)

```typescript
// Cron job runs every minute
async function evaluateAlerts() {
  const rules = await prisma.alertRule.findMany({
    where: { enabled: true }
  });

  for (const rule of rules) {
    const isTriggered = await checkAlertCondition(rule);

    if (isTriggered) {
      // Check if already active
      const activeAlert = await prisma.alertHistory.findFirst({
        where: {
          ruleId: rule.id,
          resolvedAt: null
        }
      });

      if (!activeAlert) {
        // Trigger new alert
        await prisma.alertHistory.create({
          data: { ruleId: rule.id }
        });

        // Send notifications
        await sendAlertNotifications(rule);
      }
    }
  }
}

async function checkAlertCondition(rule: AlertRule): Promise<boolean> {
  const { metric, operator, threshold, duration } = rule.condition;

  // Get metric value from last `duration` seconds
  const since = new Date(Date.now() - duration * 1000);

  const value = await getMetricValue(metric, since);

  // Evaluate condition
  switch (operator) {
    case ">": return value > threshold;
    case "<": return value < threshold;
    case ">=": return value >= threshold;
    case "<=": return value <= threshold;
    default: return false;
  }
}
```
